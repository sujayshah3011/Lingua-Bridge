BLOCK 1 — UNDERSTAND THE ENVIRONMENT PROBLEM
════════════════════════════════════════════════════

There are THREE runtime environments this app runs in:

  ENV 1: Replit web preview (browser tab)
    → Platform.OS === 'web'
    → Firebase signInWithPopup works perfectly
    → Native modules NOT available

  ENV 2: Expo Go / Rork mobile preview  
    → Platform.OS === 'android' but NO native modules
    → @react-native-google-signin CRASHES here
    → expo-auth-session works but has redirect issues
    → This is NOT a valid test env for Google Sign-In

  ENV 3: EAS Development Build / Production APK  ← TARGET
    → Real Android binary with native modules compiled in
    → @react-native-google-signin works perfectly here
    → This is what gets published to Play Store

The strategy:
  - Web preview  → Firebase signInWithPopup (already working ✓)
  - Production   → @react-native-google-signin native SDK
  - Expo Go      → Show "Use production build" message (not worth fixing)

BLOCK 2 — IMPLEMENT FINAL GOOGLE SIGN-IN ARCHITECTURE
════════════════════════════════════════════════════

Replace lib/auth.ts entirely with this platform-aware implementation:

  import { Platform } from 'react-native';
  import {
    GoogleAuthProvider,
    signInWithPopup,
    signInWithCredential,
    signOut as firebaseSignOut,
  } from 'firebase/auth';
  import {
    doc, getDoc, setDoc, updateDoc,
    addDoc, collection, serverTimestamp, increment,
  } from 'firebase/firestore';
  import { auth, db } from './firebase';
  import { WELCOME_CREDIT_INR } from './constants';

  // ─── Lazy-load native Google Sign-In only on real native builds ──
  // This prevents the RNGoogleSignin crash in Expo Go
  let GoogleSignin: any = null;
  let statusCodes: any = null;

  const loadNativeGoogleSignin = () => {
    if (GoogleSignin) return true;
    try {
      const module = require('@react-native-google-signin/google-signin');
      GoogleSignin = module.GoogleSignin;
      statusCodes = module.statusCodes;
      return true;
    } catch (e) {
      return false;
    }
  };

  // ─── WEB CLIENT ID (used by both web popup and native SDK) ───────
  const WEB_CLIENT_ID =
    '<YOUR_WEB_OAUTH_CLIENT_ID>.apps.googleusercontent.com';

  // ─── INIT (call once at app startup in App.tsx) ───────────────────
  export const initGoogleSignIn = () => {
    if (Platform.OS === 'web') return; // web uses popup, no init needed
    const loaded = loadNativeGoogleSignin();
    if (loaded && GoogleSignin) {
      GoogleSignin.configure({
        webClientId: WEB_CLIENT_ID,
        offlineAccess: true,
      });
    }
  };

  // ─── MAIN SIGN-IN FUNCTION ────────────────────────────────────────
  export const signInWithGoogle = async (): Promise<boolean> => {
    if (Platform.OS === 'web') {
      return signInWithGoogleWeb();
    } else {
      return signInWithGoogleNative();
    }
  };

  // ─── WEB: Firebase popup (works in Replit preview) ───────────────
  const signInWithGoogleWeb = async (): Promise<boolean> => {
    try {
      const provider = new GoogleAuthProvider();
      provider.addScope('profile');
      provider.addScope('email');
      const result = await signInWithPopup(auth, provider);
      const isNew = result.additionalUserInfo?.isNewUser ?? false;
      await handlePostSignIn(result.user, isNew);
      return true;
    } catch (err: any) {
      if (
        err.code === 'auth/popup-closed-by-user' ||
        err.code === 'auth/cancelled-popup-request'
      ) return false;
      throw err;
    }
  };

  // ─── NATIVE: @react-native-google-signin (works in EAS build) ────
  const signInWithGoogleNative = async (): Promise<boolean> => {
    const loaded = loadNativeGoogleSignin();

    // Expo Go / Rork preview: native module not available
    if (!loaded || !GoogleSignin) {
      throw new Error('EXPO_GO_LIMITATION');
    }

    try {
      await GoogleSignin.hasPlayServices({ showPlayServicesUpdateDialog: true });
      // Force account picker every time
      try { await GoogleSignin.signOut(); } catch (_) {}

      await GoogleSignin.signIn();
      const { idToken } = await GoogleSignin.getTokens();

      if (!idToken) throw new Error('NO_ID_TOKEN');

      const credential = GoogleAuthProvider.credential(idToken);
      const result = await signInWithCredential(auth, credential);

      const snap = await getDoc(doc(db, 'users', result.user.uid));
      const isNew = !snap.exists();
      await handlePostSignIn(result.user, isNew);
      return true;
    } catch (err: any) {
      if (err.code === statusCodes?.SIGN_IN_CANCELLED) return false;
      if (err.code === statusCodes?.IN_PROGRESS) return false;
      throw err;
    }
  };

  // ─── SHARED: Create/update user in Firestore ─────────────────────
  const handlePostSignIn = async (user: any, isNew: boolean) => {
    const userRef = doc(db, 'users', user.uid);

    if (isNew) {
      const snap = await getDoc(userRef);
      if (!snap.exists()) {
        await setDoc(userRef, {
          uid:           user.uid,
          email:         user.email,
          displayName:   user.displayName,
          photoURL:      user.photoURL,
          createdAt:     serverTimestamp(),
          credits:       WELCOME_CREDIT_INR,
          totalCredited: WELCOME_CREDIT_INR,
          totalDebited:  0,
          stats: {
            charsTranslated: 0,
            charsTTS:        0,
            secondsSTT:      0,
            sessionCount:    1,
          },
          plan: 'pay-as-you-go',
          isActive: true,
        });
        await addDoc(collection(db, 'transactions'), {
          uid:           user.uid,
          type:          'topup',
          chars:         null,
          seconds:       null,
          userCharged:   -WELCOME_CREDIT_INR,
          balanceBefore: 0,
          balanceAfter:  WELCOME_CREDIT_INR,
          note:          'Welcome credit',
          success:       true,
          timestamp:     serverTimestamp(),
        });
      } else {
        await updateDoc(userRef, { 'stats.sessionCount': increment(1) });
      }
    } else {
      try {
        await updateDoc(userRef, { 'stats.sessionCount': increment(1) });
      } catch (_) {}
    }
  };

  // ─── SIGN OUT ─────────────────────────────────────────────────────
  export const signOut = async () => {
    if (Platform.OS !== 'web') {
      try {
        const loaded = loadNativeGoogleSignin();
        if (loaded && GoogleSignin) await GoogleSignin.signOut();
      } catch (_) {}
    }
    await firebaseSignOut(auth);
  };

  # BhaashAI — Indian Language Translator

  - Node.js 18+

```bash
  cp .env.example .env
```

```bash
  npm run start:web
```

```bash
  # One-time: log in to Expo
  eas login

  # Build development APK (includes native Google Sign-In)
  npm run build:dev

  # Download the APK from the link EAS provides
```
