  import {

  // ─── SIGN IN WITH EMAIL ──────────────────────────────────────────
  export const signInWithEmail = async (
    email: string,
    password: string
  ): Promise<boolean> => {
    try {
      const result = await signInWithEmailAndPassword(auth, email, password);
      
      // Check if user doc exists in Firestore
      const userRef = doc(db, 'users', result.user.uid);
      const snap = await getDoc(userRef);
      
      if (!snap.exists()) {
        // First time signing in with email (user created via Firebase Console)
        // Create their Firestore document with ₹100 welcome credit
        await setDoc(userRef, {
          uid:           result.user.uid,
          email:         result.user.email,
          displayName:   result.user.displayName ?? email.split('@')[0],
          photoURL:      result.user.photoURL ?? null,
          createdAt:     serverTimestamp(),
          credits:       WELCOME_CREDIT_INR,
          totalCredited: WELCOME_CREDIT_INR,
          totalDebited:  0,
          stats: {
            charsTranslated: 0,
            charsTTS:        0,
            secondsSTT:      0,
            sessionCount:    1,
          },
          plan:     'pay-as-you-go',
          isActive: true,
        });
        await addDoc(collection(db, 'transactions'), {
          uid:           result.user.uid,
          type:          'topup',
          chars:         null,
          seconds:       null,
          userCharged:   -WELCOME_CREDIT_INR,
          balanceBefore: 0,
          balanceAfter:  WELCOME_CREDIT_INR,
          note:          'Welcome credit',
          success:       true,
          timestamp:     serverTimestamp(),
        });
      } else {
        // Existing user — increment session count
        await updateDoc(userRef, {
          'stats.sessionCount': increment(1),
        });
      }
      return true;
    } catch (err: any) {
      // Translate Firebase error codes to human-readable messages
      throw translateAuthError(err);
    }
  };

  // ─── SIGN UP WITH EMAIL ──────────────────────────────────────────
  export const signUpWithEmail = async (
    email: string,
    password: string,
    displayName: string
  ): Promise<boolean> => {
    try {
      const result = await createUserWithEmailAndPassword(auth, email, password);
      
      // Update Firebase Auth profile with display name
      await updateProfile(result.user, { displayName });
      
      // Create Firestore user document with ₹100 welcome credit
      await setDoc(doc(db, 'users', result.user.uid), {
        uid:           result.user.uid,
        email:         result.user.email,
        displayName,
        photoURL:      null,
        createdAt:     serverTimestamp(),
        credits:       WELCOME_CREDIT_INR,
        totalCredited: WELCOME_CREDIT_INR,
        totalDebited:  0,
        stats: {
          charsTranslated: 0,
          charsTTS:        0,
          secondsSTT:      0,
          sessionCount:    1,
        },
        plan:     'pay-as-you-go',
        isActive: true,
      });
      await addDoc(collection(db, 'transactions'), {
        uid:           result.user.uid,
        type:          'topup',
        chars:         null,
        seconds:       null,
        userCharged:   -WELCOME_CREDIT_INR,
        balanceBefore: 0,
        balanceAfter:  WELCOME_CREDIT_INR,
        note:          'Welcome credit',
        success:       true,
        timestamp:     serverTimestamp(),
      });
      return true;
    } catch (err: any) {
      throw translateAuthError(err);
    }
  };

  // ─── FORGOT PASSWORD ─────────────────────────────────────────────
  export const sendPasswordReset = async (email: string): Promise<void> => {
    try {
      await sendPasswordResetEmail(auth, email);
    } catch (err: any) {
      throw translateAuthError(err);
    }
  };

  // ─── ERROR TRANSLATOR ────────────────────────────────────────────
  const translateAuthError = (err: any): Error => {
    const messages: Record<string, string> = {
      'auth/invalid-email':          'Invalid email address.',
      'auth/user-not-found':         'No account found with this email.',
      'auth/wrong-password':         'Incorrect password. Try again.',
      'auth/invalid-credential':     'Incorrect email or password.',
      'auth/email-already-in-use':   'An account with this email already exists.',
      'auth/weak-password':          'Password must be at least 6 characters.',
      'auth/too-many-requests':      'Too many attempts. Try again later.',
      'auth/network-request-failed': 'Network error. Check your connection.',
      'auth/user-disabled':          'This account has been disabled.',
    };
    const message = messages[err.code] ?? err.message ?? 'Authentication failed.';
    return new Error(message);
  };
