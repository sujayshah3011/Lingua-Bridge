SECTION 3 — CREDIT ENGINE (lib/credits.ts)
════════════════════════════════════════════

Create lib/credits.ts with these exact functions:

  import { db } from './firebase';
  import { 
    doc, getDoc, updateDoc, addDoc, 
    collection, increment, serverTimestamp, runTransaction 
  } from 'firebase/firestore';
  import { USER_PRICE, SARVAM_COST } from './constants';

  // ─── CHECK BALANCE ────────────────────────────────────────────────
  export const getBalance = async (uid: string): Promise<number> => {
    const snap = await getDoc(doc(db, 'users', uid));
    return snap.data()?.credits ?? 0;
  };

  export const hasEnoughCredits = async (
    uid: string, 
    estimatedCost: number
  ): Promise<boolean> => {
    const balance = await getBalance(uid);
    return balance > estimatedCost;
  };

  // ─── CORE DEDUCT FUNCTION ─────────────────────────────────────────
  // Uses Firestore transaction to prevent race conditions
  const deductCredits = async ({
    uid,
    type,
    chars = null,
    seconds = null,
    userCharged,
    sarvamCost,
    sourceLang = null,
    targetLang = null,
  }: {
    uid: string;
    type: string;
    chars?: number | null;
    seconds?: number | null;
    userCharged: number;
    sarvamCost: number;
    sourceLang?: string | null;
    targetLang?: string | null;
  }) => {
    const userRef = doc(db, 'users', uid);
    
    await runTransaction(db, async (txn) => {
      const userSnap = await txn.get(userRef);
      if (!userSnap.exists()) throw new Error('User not found');
      
      const balanceBefore = userSnap.data().credits;
      const balanceAfter = parseFloat((balanceBefore - userCharged).toFixed(6));
      
      // Update user document
      txn.update(userRef, {
        credits: balanceAfter,
        totalDebited: increment(userCharged),
      });
      
      // Write transaction record
      const txnRef = doc(collection(db, 'transactions'));
      txn.set(txnRef, {
        uid,
        type,
        chars,
        seconds,
        sarvamCost: parseFloat(sarvamCost.toFixed(6)),
        userCharged: parseFloat(userCharged.toFixed(6)),
        balanceBefore: parseFloat(balanceBefore.toFixed(6)),
        balanceAfter,
        sourceLang,
        targetLang,
        success: true,
        timestamp: serverTimestamp(),
      });
    });
  };

  // ─── UPDATE STATS ─────────────────────────────────────────────────
  const updateStats = async (uid: string, statsUpdate: Record<string, any>) => {
    await updateDoc(doc(db, 'users', uid), statsUpdate);
  };

  // ─── PUBLIC API: CHARGE FOR TRANSLATE ────────────────────────────
  export const chargeTranslate = async (
    uid: string,
    inputText: string,
    sourceLang: string,
    targetLang: string,
  ) => {
    const chars = inputText.length;
    const sarvamCost = chars * SARVAM_COST.translate.perChar;
    const userCharged = chars * USER_PRICE.translate.perChar;
    
    await deductCredits({ uid, type: 'translate', chars, sarvamCost, userCharged, sourceLang, targetLang });
    await updateStats(uid, { 'stats.charsTranslated': increment(chars) });
    
    return { chars, cost: userCharged };
  };

  // ─── PUBLIC API: CHARGE FOR TTS ──────────────────────────────────
  export const chargeTTS = async (
    uid: string,
    text: string,
    targetLang: string,
  ) => {
    const chars = text.length;
    const sarvamCost = chars * SARVAM_COST.tts.perChar;
    const userCharged = chars * USER_PRICE.tts.perChar;
    
    await deductCredits({ uid, type: 'tts', chars, sarvamCost, userCharged, targetLang });
    await updateStats(uid, { 'stats.charsTTS': increment(chars) });
    
    return { chars, cost: userCharged };
  };

  // ─── PUBLIC API: CHARGE FOR STT ──────────────────────────────────
  export const chargeSTT = async (
    uid: string,
    audioSeconds: number,
    sourceLang: string,
  ) => {
    const seconds = parseFloat(audioSeconds.toFixed(2));
    const sarvamCost = seconds * SARVAM_COST.stt.perSecond;
    const userCharged = seconds * USER_PRICE.stt.perSecond;
    
    await deductCredits({ uid, type: 'stt', seconds, sarvamCost, userCharged, sourceLang });
    await updateStats(uid, { 'stats.secondsSTT': increment(seconds) });
    
    return { seconds, cost: userCharged };
  };

  // ─── PUBLIC API: CHARGE FOR TRANSLITERATE ────────────────────────
  export const chargeTransliterate = async (uid: string, text: string) => {
    const chars = text.length;
    const sarvamCost = chars * SARVAM_COST.transliterate.perChar;
    const userCharged = chars * USER_PRICE.transliterate.perChar;
    
    await deductCredits({ uid, type: 'transliterate', chars, sarvamCost, userCharged });
    await updateStats(uid, { 'stats.charsTransliterated': increment(chars) });
    
    return { chars, cost: userCharged };
  };

  // ─── PRE-FLIGHT CHECK + COST ESTIMATE ────────────────────────────
  // Call this BEFORE making any API call to show user estimated cost
  export const estimateCost = (
    type: 'translate' | 'tts' | 'stt' | 'transliterate',
    quantity: number  // chars for text, seconds for stt
  ): number => {
    switch (type) {
      case 'translate':    return quantity * USER_PRICE.translate.perChar;
      case 'tts':          return quantity * USER_PRICE.tts.perChar;
      case 'stt':          return quantity * USER_PRICE.stt.perSecond;
      case 'transliterate': return quantity * USER_PRICE.transliterate.perChar;
      default: return 0;
    }
  };

  export const callSarvamTranslate = async (
    uid: string,
    text: string, 
    sourceLang: string, 
    targetLang: string,
  ) => {
    // 1. Estimate cost
    const estimatedCost = estimateCost('translate', text.length);
    
    // 2. Check balance
    const canProceed = await hasEnoughCredits(uid, estimatedCost);
    if (!canProceed) throw new Error('INSUFFICIENT_CREDITS');
    
    // 3. Call Sarvam API
    const response = await fetch('https://api.sarvam.ai/translate', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'api-subscription-key': SARVAM_API_KEY,
      },
      body: JSON.stringify({
        input: text,
        source_language_code: sourceLang,
        target_language_code: targetLang,
        model: 'sarvam-translate:v1',
        enable_preprocessing: true,
      }),
    });
    
    if (!response.ok) throw new Error(`TRANSLATE_ERROR_${response.status}`);
    const data = await response.json();
    
    // 4. Charge AFTER successful response only
    const actualChars = text.length;  // charge on input chars
    await chargeTranslate(uid, text, sourceLang, targetLang);
    
    return data.translated_text;
  };

  // stt.ts — wrap transcribeAudio, charge AFTER audio stops:
  // Track recording start time with: const startTime = Date.now();
  // After STT returns: const audioSeconds = (Date.now() - startTime) / 1000;

  // tts.ts — wrap synthesizeSpeech:
  // After successful TTS: await chargeTTS(uid, text, targetLang);

  // CRITICAL RULE: NEVER charge if API call fails.
  // Only deduct credits on HTTP 200 success responses.

SECTION 11 — INSUFFICIENT CREDITS HANDLER
════════════════════════════════════════════

Create a reusable hook: hooks/useCredits.ts

  export const useCredits = (uid: string) => {
    const [credits, setCredits] = useState(0);
    const [loading, setLoading] = useState(true);
    
    useEffect(() => {
      const unsub = onSnapshot(doc(db, 'users', uid), (snap) => {
        setCredits(snap.data()?.credits ?? 0);
        setLoading(false);
      });
      return unsub;
    }, [uid]);
    
    const requireCredits = useCallback(async (
      estimatedCost: number,
      onInsufficent: () => void
    ) => {
      if (credits <= estimatedCost) {
        onInsufficent();
        return false;
      }
      return true;
    }, [credits]);
    
    return { credits, loading, requireCredits };
  };

Use in every screen that calls APIs:
  const { credits, requireCredits } = useCredits(user.uid);
  
  const handleTranslate = async () => {
    const est = estimateCost('translate', inputText.length);
    const ok = await requireCredits(est, () => navigate('TopUp'));
    if (!ok) return;
    // proceed with translation
  };
