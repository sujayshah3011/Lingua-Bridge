BLOCK 13 — FIRESTORE SECURITY RULES VERIFICATION
════════════════════════════════════════════════════

CHECK 13.1 — Rules must be set to this exact content
  Verify (or instruct user to set) in Firebase Console → Firestore → Rules:

  rules_version = '2';
  service cloud.firestore {
    match /databases/{database}/documents {
      
      // Users can only read/write their OWN document
      match /users/{uid} {
        allow read, write: if request.auth != null && request.auth.uid == uid;
      }
      
      // Users can only read their OWN transactions
      match /transactions/{txnId} {
        allow read: if request.auth != null && 
                       resource.data.uid == request.auth.uid;
        // Only server (app code) writes transactions — 
        // users cannot write directly
        allow write: if request.auth != null;
      }
      
      // Topups: users can read own, not write directly
      match /topups/{topupId} {
        allow read: if request.auth != null && 
                       resource.data.uid == request.auth.uid;
        allow write: if request.auth != null;
      }
    }
  }

  ✓ No rule allows unauthenticated access to any collection
  ✓ Users cannot read other users' documents
