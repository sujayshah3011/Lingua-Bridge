Create lib/credits.ts with atomic Firestore transactions:

  import { db } from './firebase';
  import {
    doc, runTransaction, addDoc, collection,
    serverTimestamp, increment
  } from 'firebase/firestore';
  import { USER_PRICE, SARVAM_COST } from './constants';

  // ─── CORE ATOMIC DEDUCT ───────────────────────────────────────────
  const atomicDeduct = async ({
    uid, type, chars = null, seconds = null,
    userCharged, sourceLang = null, targetLang = null,
  }: {
    uid: string; type: string;
    chars?: number | null; seconds?: number | null;
    userCharged: number;
    sourceLang?: string | null; targetLang?: string | null;
  }) => {
    const userRef = doc(db, 'users', uid);

    let balanceBefore = 0;
    let balanceAfter = 0;

    await runTransaction(db, async (txn) => {
      const snap = await txn.get(userRef);
      if (!snap.exists()) throw new Error('USER_NOT_FOUND');
      
      balanceBefore = snap.data().credits;
      
      // BLOCK if insufficient — never go negative
      if (balanceBefore < userCharged) {
        throw new Error('INSUFFICIENT_CREDITS');
      }
      
      // Round to 6 decimal places to avoid floating point drift
      balanceAfter = Math.round((balanceBefore - userCharged) * 1000000) / 1000000;
      
      txn.update(userRef, {
        credits: balanceAfter,
        totalDebited: increment(userCharged),
      });
    });

    // Log transaction AFTER successful balance update
    await addDoc(collection(db, 'transactions'), {
      uid, type, chars, seconds,
      userCharged: Math.round(userCharged * 1000000) / 1000000,
      balanceBefore: Math.round(balanceBefore * 1000000) / 1000000,
      balanceAfter:  Math.round(balanceAfter  * 1000000) / 1000000,
      sourceLang, targetLang,
      success: true,
      timestamp: serverTimestamp(),
    });
  };

  // ─── TRANSLATE ────────────────────────────────────────────────────
  export const chargeTranslate = async (
    uid: string, text: string,
    sourceLang: string, targetLang: string
  ) => {
    const chars = text.length;
    const userCharged = chars * USER_PRICE.translate.perChar;
    await atomicDeduct({ uid, type: 'translate', chars, userCharged, sourceLang, targetLang });
    // Update stats
    await import('firebase/firestore').then(({ updateDoc, doc: fdoc, increment: inc }) =>
      updateDoc(fdoc(db, 'users', uid), { 'stats.charsTranslated': inc(chars) })
    );
    return { chars, cost: userCharged };
  };

  // ─── TTS ──────────────────────────────────────────────────────────
  export const chargeTTS = async (
    uid: string, text: string, targetLang: string
  ) => {
    const chars = text.length;
    const userCharged = chars * USER_PRICE.tts.perChar;
    await atomicDeduct({ uid, type: 'tts', chars, userCharged, targetLang });
    await import('firebase/firestore').then(({ updateDoc, doc: fdoc, increment: inc }) =>
      updateDoc(fdoc(db, 'users', uid), { 'stats.charsTTS': inc(chars) })
    );
    return { chars, cost: userCharged };
  };

  // ─── STT ──────────────────────────────────────────────────────────
  export const chargeSTT = async (
    uid: string, audioSeconds: number, sourceLang: string
  ) => {
    const seconds = Math.round(audioSeconds * 100) / 100;
    const userCharged = seconds * USER_PRICE.stt.perSecond;
    await atomicDeduct({ uid, type: 'stt', seconds, userCharged, sourceLang });
    await import('firebase/firestore').then(({ updateDoc, doc: fdoc, increment: inc }) =>
      updateDoc(fdoc(db, 'users', uid), { 'stats.secondsSTT': inc(seconds) })
    );
    return { seconds, cost: userCharged };
  };

  // ─── TRANSLITERATE ────────────────────────────────────────────────
  export const chargeTransliterate = async (uid: string, text: string) => {
    const chars = text.length;
    const userCharged = chars * USER_PRICE.transliterate.perChar;
    await atomicDeduct({ uid, type: 'transliterate', chars, userCharged });
    return { chars, cost: userCharged };
  };

  // ─── COST ESTIMATOR (pre-flight UI display) ───────────────────────
  export const estimateCost = (
    type: 'translate' | 'tts' | 'stt' | 'transliterate',
    quantity: number
  ): number => {
    const rates = {
      translate:     USER_PRICE.translate.perChar,
      tts:           USER_PRICE.tts.perChar,
      stt:           USER_PRICE.stt.perSecond,
      transliterate: USER_PRICE.transliterate.perChar,
    };
    return quantity * rates[type];
  };

  // ─── BALANCE GUARD (call before every API action) ─────────────────
  export const assertSufficientCredits = async (
    uid: string,
    type: 'translate' | 'tts' | 'stt' | 'transliterate',
    quantity: number
  ): Promise<void> => {
    const { getDoc, doc: fdoc } = await import('firebase/firestore');
    const snap = await getDoc(fdoc(db, 'users', uid));
    const balance = snap.data()?.credits ?? 0;
    const needed = estimateCost(type, quantity);
    if (balance < needed) {
      throw new Error('INSUFFICIENT_CREDITS');
    }
  };
